cmake_minimum_required(VERSION 3.20.0)
project(ulf-validate3)
set(PLATFORM "POSIX" CACHE STRING "")
set(NETWORK_CHANNEL_TCP_POSIX ON CACHE STRING "")
set(FEDERATED ON CACHE STRING "")
add_subdirectory($ENV{REACTOR_UC_PATH} reactor-uc)
target_compile_definitions(reactor-uc PUBLIC LF_LOG_LEVEL_ALL=LF_LOG_LEVEL_WARN)
add_executable(sensor_a sensor.c)
target_compile_definitions(sensor_a PRIVATE NAME="sensor A" PERIOD_MS=100 PORT_NUM=8921)
target_link_libraries(sensor_a PRIVATE reactor-uc)
add_executable(sensor_b sensor.c)
target_compile_definitions(sensor_b PRIVATE NAME="sensor B" PERIOD_MS=150 PORT_NUM=8922)
target_link_libraries(sensor_b PRIVATE reactor-uc)
add_executable(sensor_b_slow sensor.c)
target_compile_definitions(sensor_b_slow PRIVATE NAME="sensor B" PERIOD_MS=150 PORT_NUM=8922 SLOW_AT=3 SLOW_MS=60)
target_link_libraries(sensor_b_slow PRIVATE reactor-uc)
add_executable(fusion_w30 fusion.c)
target_compile_definitions(fusion_w30 PRIVATE MAXWAIT_MS=30)
target_link_libraries(fusion_w30 PRIVATE reactor-uc)
add_executable(fusion_w100 fusion.c)
target_compile_definitions(fusion_w100 PRIVATE MAXWAIT_MS=100)
target_link_libraries(fusion_w100 PRIVATE reactor-uc)
# W sweep: one fusion executable per waiting bound
foreach(W 55 59 60 61 62 65 70)
  add_executable(fusion_w${W} fusion.c)
  target_compile_definitions(fusion_w${W} PRIVATE MAXWAIT_MS=${W})
  target_link_libraries(fusion_w${W} PRIVATE reactor-uc)
endforeach()
