cmake_minimum_required(VERSION 3.20.0)
project(ulf-tardy)
set(PLATFORM "POSIX" CACHE STRING "")
set(NETWORK_CHANNEL_TCP_POSIX ON CACHE STRING "")
set(FEDERATED ON CACHE STRING "")
add_subdirectory($ENV{REACTOR_UC_PATH} reactor-uc)
target_compile_definitions(reactor-uc PUBLIC LF_LOG_LEVEL_ALL=LF_LOG_LEVEL_WARN)
add_executable(src src.c)
target_link_libraries(src PRIVATE reactor-uc)
add_executable(dst_short dst.c)
target_compile_definitions(dst_short PRIVATE MAXWAIT_MS=50)
target_link_libraries(dst_short PRIVATE reactor-uc)
add_executable(dst_long dst.c)
target_compile_definitions(dst_long PRIVATE MAXWAIT_MS=150)
target_link_libraries(dst_long PRIVATE reactor-uc)
add_executable(dst_nohandler dst.c)
target_compile_definitions(dst_nohandler PRIVATE MAXWAIT_MS=50 NO_HANDLER)
target_link_libraries(dst_nohandler PRIVATE reactor-uc)
add_executable(src_sparse src.c)
target_compile_definitions(src_sparse PRIVATE SPARSE)
target_link_libraries(src_sparse PRIVATE reactor-uc)
